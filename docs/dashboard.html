Submission history

The behind-the-scenes story of how papers get published.

submissions = FileAttachment("data/submissions.json").json()
papers = {
  const groups = new Map();
  for (const s of submissions) {
    if (!groups.has(s.paper)) groups.set(s.paper, []);
    groups.get(s.paper).push(s);
  }
  return [...groups.entries()];
}

sortedPapers = papers.map(([name, subs]) => {
  const sorted = [...subs].sort((a, b) => {
    if (!a.submission_date && !b.submission_date) return 0;
    if (!a.submission_date) return 1;
    if (!b.submission_date) return -1;
    return a.submission_date.localeCompare(b.submission_date);
  });
  return [name, sorted];
});

orderedPapers = [...sortedPapers].sort(([, a], [, b]) => {
  const ad = a.find(s => s.submission_date)?.submission_date || "0000";
  const bd = b.find(s => s.submission_date)?.submission_date || "0000";
  return bd.localeCompare(ad);
});
function daysBetween(d1, d2) {
  if (!d1 || !d2) return null;
  return Math.round((new Date(d2) - new Date(d1)) / 86400000);
}

function fmtDate(d) {
  if (!d) return null;
  return new Date(d + "T12:00:00").toLocaleDateString("en-US", { month: "short", year: "numeric" });
}

// Abbreviate journal names for chart labels
function abbrevJournal(name) {
  const stop = new Set(["the", "of", "and", "&", "in", "for", "a", "an"]);
  const words = name.split(/\s+/).filter(w => !stop.has(w.toLowerCase()));
  if (words.length === 1) return words[0];
  if (words.length === 2)
    return words[0] + (words[1].length > 7 ? " " + words[1].slice(0, 5) + "." : " " + words[1]);
  // 3+ words: first word + initials of rest
  return words[0] + " " + words.slice(1).map(w => w[0]).join("").toUpperCase();
}

// Wrap a title string at a word boundary, inserting \n after the first line
function wrapTitle(title, maxLen) {
  if (title.length <= maxLen) return title;
  const words = title.split(" ");
  let line1 = "";
  for (const w of words) {
    const next = line1 ? line1 + " " + w : w;
    if (next.length > maxLen && line1) break;
    line1 = next;
  }
  const line2 = title.slice(line1.length + 1);
  return line2 ? line1 + "\n" + line2 : line1;
}

stats = {
  const total = orderedPapers.length;
  const firstAuthor = orderedPapers.filter(([, s]) => s.some(r => r.authorship === "first")).length;

  const acceptedPapers = orderedPapers.filter(([, s]) =>
    s.some(r => r.status === "accepted" || r.status === "published")
  );

  // Journal rejection rate: per unique (paper × journal) stint, % that ended in rejection.
  // Liz Stuart benchmark: aim for ≥60% to ensure you're submitting ambitiously.
  const pjGroups = new Map();
  for (const s of submissions) {
    const key = `${s.paper}|||${s.journal}`;
    if (!pjGroups.has(key)) pjGroups.set(key, []);
    pjGroups.get(key).push(s);
  }
  const pjOutcomes = [...pjGroups.values()].map(rows => {
    const statuses = rows.map(r => r.status);
    if (statuses.some(s => s === "accepted" || s === "published")) return "accepted";
    if (statuses.some(s => s === "desk rejected" || s === "peer review rejected")) return "rejected";
    return "pending";
  });
  const completedPJ = pjOutcomes.filter(s => s !== "pending");
  const rejectedPJ = pjOutcomes.filter(s => s === "rejected");
  const rejectionRate = completedPJ.length > 0
    ? Math.round(100 * rejectedPJ.length / completedPJ.length) : null;

  // Journal stints per accepted paper (paper-journal pairs attempted before final acceptance)
  const avgStints = acceptedPapers.length > 0 ? (() => {
    const counts = acceptedPapers.map(([, subs]) => {
      const journals = new Set(subs.map(s => s.journal));
      return journals.size;
    });
    return (counts.reduce((a, b) => a + b, 0) / counts.length).toFixed(1);
  })() : null;

  const deskRows = submissions.filter(s =>
    s.status === "desk rejected" && s.submission_date && s.decision_date
  );
  const avgDesk = deskRows.length > 0
    ? Math.round(deskRows.reduce((n, s) => n + daysBetween(s.submission_date, s.decision_date), 0) / deskRows.length)
    : null;

  const ttAccept = acceptedPapers.map(([, subs]) => {
    const first = subs.find(s => s.submission_date);
    const acc = subs.find(s => s.status === "accepted" || s.status === "published");
    if (!first || !acc?.decision_date) return null;
    return daysBetween(first.submission_date, acc.decision_date);
  }).filter(d => d !== null);
  const avgTTA = ttAccept.length > 0
    ? Math.round(ttAccept.reduce((a, b) => a + b, 0) / ttAccept.length) : null;

  return { total, firstAuthor, acceptedCount: acceptedPapers.length,
           rejectionRate, avgStints, avgDesk, avgTTA };
}
// Journal statistics (paper-level: one row per paper-journal pair)
journalStats = {
  // Group by (paper, journal)
  const pjMap = new Map();
  for (const s of submissions) {
    const key = `${s.paper}|||${s.journal}`;
    if (!pjMap.has(key)) pjMap.set(key, { paper: s.paper, journal: s.journal, rows: [] });
    pjMap.get(key).rows.push(s);
  }

  // Summarize each paper-journal pair
  const pjSummaries = [...pjMap.values()].map(({ paper, journal, rows }) => {
    const accepted = rows.some(r => r.status === "accepted" || r.status === "published");
    const revisions = rows.filter(r => r.status === "revise").length;
    // Days from first submission at this journal to final decision at this journal
    const subDates = rows.filter(r => r.submission_date).map(r => r.submission_date).sort();
    const decDates = rows.filter(r => r.decision_date && r.status !== "pending").map(r => r.decision_date).sort();
    const days = subDates.length > 0 && decDates.length > 0
      ? daysBetween(subDates[0], decDates[decDates.length - 1]) : null;
    return { paper, journal, accepted, revisions, days };
  });

  // Roll up to journal level
  const byJournal = new Map();
  for (const pj of pjSummaries) {
    if (!byJournal.has(pj.journal)) byJournal.set(pj.journal, []);
    byJournal.get(pj.journal).push(pj);
  }

  return [...byJournal.entries()].map(([journal, pjs]) => {
    const nPapers = pjs.length;
    const nAccepted = pjs.filter(p => p.accepted).length;
    const rate = Math.round(100 * nAccepted / nPapers);
    const withRevisions = pjs.filter(p => p.revisions > 0);
    const avgRevisions = withRevisions.length > 0
      ? (withRevisions.reduce((n, p) => n + p.revisions, 0) / withRevisions.length).toFixed(1)
      : "—";
    const withDays = pjs.filter(p => p.days !== null);
    const avgDays = withDays.length > 0
      ? Math.round(withDays.reduce((n, p) => n + p.days, 0) / withDays.length) : null;
    return { journal, nPapers, nAccepted, rate, avgRevisions, avgDays };
  }).sort((a, b) => b.nPapers - a.nPapers || b.rate - a.rate);
}

Paper journeys

{
  if (orderedPapers.length === 0) return html`<p class="db-empty">No submissions tracked yet.</p>`;

  const paperNames = [...orderedPapers.map(([name]) => name)].reverse();

  // Journey line per paper: first submission_date → last decision_date
  const lines = orderedPapers.flatMap(([paper, subs]) => {
    const allDates = [
      ...subs.filter(s => s.submission_date).map(s => new Date(s.submission_date + "T12:00:00")),
      ...subs.filter(s => s.decision_date).map(s => new Date(s.decision_date + "T12:00:00"))
    ];
    return allDates.length >= 2
      ? [{ paper, start: d3.min(allDates), end: d3.max(allDates) }] : [];
  });

  // Decision dots: one per submission row at decision_date
  const events = submissions
    .filter(s => s.decision_date)
    .map(s => ({
      paper: s.paper,
      date: new Date(s.decision_date + "T12:00:00"),
      status: s.status,
      journal: s.journal,
      label: `${s.journal} — ${STATUS_LABEL[s.status] || s.status} (${fmtDate(s.decision_date)})`
    }));

  // Journal segment starts: first submission_date per (paper, journal), for label anchors
  const segStarts = (() => {
    const seen = new Map();
    for (const s of submissions) {
      const key = `${s.paper}|||${s.journal}`;
      if (!seen.has(key) && s.submission_date)
        seen.set(key, { paper: s.paper, journal: s.journal, date: new Date(s.submission_date + "T12:00:00") });
    }
    return [...seen.values()];
  })();

  return Plot.plot({
    marginLeft: 175,
    marginRight: 20,
    marginTop: 20,
    marginBottom: 40,
    height: Math.max(200, orderedPapers.length * 85 + 80),
    x: { type: "time", label: null, tickFormat: "%b %Y" },
    y: { label: null, domain: paperNames, tickFormat: d => wrapTitle(d, 18) },
    color: {
      domain: ["desk rejected", "revise", "accepted", "published", "peer review rejected", "pending"],
      range: ["#9a3412", "#1e40af", "#166534", "#14532d", "#92400e", "#9ca3af"],
      legend: true
    },
    marks: [
      // Journey line
      Plot.link(lines, {
        x1: "start", x2: "end", y1: "paper", y2: "paper",
        stroke: "#d1d5db", strokeWidth: 2.5
      }),
      // Journal segment start markers (gray diamonds) — hover for journal name
      Plot.dot(segStarts, {
        x: "date", y: "paper",
        fill: "#9ca3af", r: 4, symbol: "diamond",
        title: d => `Submitted to ${d.journal}\n${fmtDate(d.date.toISOString().slice(0, 10))}`
      }),
      // Decision outcome dots — hover for full detail
      Plot.dot(events, {
        x: "date", y: "paper",
        fill: "status", stroke: "white", strokeWidth: 1.5,
        r: 8, title: "label"
      })
    ]
  });
}
